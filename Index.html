// Apni Apps Script deploy URL yaha daalo const API_URL = "https://script.google.com/macros/s/XXXXX/exec"; async function loadProducts() { const response = await fetch(API_URL); const products = await response.json(); const grid = document.querySelector(".prod-grid"); grid.innerHTML = ""; // purana hata do products.forEach(p => { if (p["Stock Status"] !== "Yes") return; // sirf jo stock me hai wo dikhao const mainImage = p["Image 1"]; const mrp = Number(p["MRP"]); const sellingPrice = Number(p["Selling Price"]); const hasDiscount = mrp > sellingPrice; // Agar discount hai to MRP crossed-out dikhayenge, warna sirf ek price const priceHTML = hasDiscount ? `₹${mrp}₹${sellingPrice}` : `₹${sellingPrice}`; grid.innerHTML += `
${p["Brand"]} · ${p["Category"]}

${p["Product Name"]} ${p["Flavour"] ? "- " + p["Flavour"] : ""}

${p["Weight/Size"] || ""}

${priceHTML}
`; }); } loadProducts();
